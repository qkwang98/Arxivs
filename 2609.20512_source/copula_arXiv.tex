\date{September 17, 2026}

\begin{abstract}
We construct a symmetric operad $\mathfrak{C}$ on the class of all multivariate copulas, with operadic composition given by Sklar substitution. This places the hierarchical combination of dependence structures into an algebraic framework. Restricting to absolutely continuous copulas whose densities lie in $L\log L$, we prove that copula entropy is additive under composition, making it an additive character on suitable finite-entropy suboperads. We exhibit explicit closed suboperads (bounded, $L^{p}$, boundary-growth) and show by counterexample that componentwise $L\log L$ does not imply closure under composition.
\end{abstract}

\maketitle

\section{Introduction}
Copulas, introduced by Sklar \cite{sklar}, encode multivariate dependence separately from marginal distributions. While extensively used in probability and statistics, the algebraic structure underlying copula composition has only recently been explored. The author \cite{lu2024} proposed the notion of a ``copula operad''; here we give a rigorous construction of the general copula operad as a symmetric operad \cite{loday} in $\mathbf{Set}$, study its absolutely continuous and finite-entropy suboperads, and characterize copula entropy as an additive character.

Our main results are as follows:
\begin{enumerate}
\item[(1)] The collection $\mathfrak{C}=\bigsqcup_{n\geq 1}\mathcal{C}_{n}$ of all $n$-copulas forms a symmetric operad under substitution (Theorem \ref{thm:operad}).
\item[(2)] The absolutely continuous subclass $\mathfrak{C}^{ac}$ is a suboperad; composition corresponds to a Jacobian density product (Proposition \ref{prop:ac}).
\item[(3)] On suboperads where all densities lie in $L\log L$, copula entropy satisfies
\[ \displaystyle H\bigl(\gamma(\Phi;\Psi_{1},\dots,\Psi_{n})\bigr)=H(\Phi)+\sum_{k=1}^{n} H(\Psi_{k}) \quad (\text{Theorem \ref{thm:add}}). \]
\item[(4)] We identify closed finite-entropy suboperads and provide a counterexample showing that $L\log L$ alone is not closed under composition (Theorem \ref{thm:counterex}).
\end{enumerate}

The paper is organized as follows. Section 2 constructs the general copula operad. Section 3 studies the absolutely continuous suboperad. Section 4 defines copula entropy and proves additivity. Section 5 exhibits closed suboperads and the non-closure counterexample. Section 6 interprets entropy as an additive character.

\section{The general copula operad}
\label{sec:operad}
Let $\I=[0,1]$. For $n\geq 1$, let $\mathcal{C}_{n}$ denote the set of $n$-copulas: functions $C:\I^{n}\to\I$ that are grounded, have uniform margins, and are $n$-increasing \cite{nelsen}.

\begin{definition}
\label{def:comp}
For $n\geq 1$, $m_{1},\dots,m_{n}\geq 1$, $\Phi\in\mathcal{C}_{n}$, $\Psi_{k}\in\mathcal{C}_{m_{k}}$, define
\[
\gamma(\Phi;\Psi_{1},\dots,\Psi_{n})(v)=\Phi\bigl(\Psi_{1}(v^{(1)}),\dots,\Psi_{n}(v^{(n)})\bigr),
\]
where $v=(v^{(1)},\dots,v^{(n)})$ with $v^{(k)}\in\I^{m_{k}}$. The unit is $\mathrm{id}\in\mathcal{C}_{1}$, $\mathrm{id}(u)=u$. For $\sigma\in S_{n}$, $(\Phi\cdot\sigma)(u_{1},\dots,u_{n})=\Phi(u_{\sigma^{-1}(1)},\dots,u_{\sigma^{-1}(n)})$.
\end{definition}

\begin{proposition}[Closure]
\label{prop:closure}
The composition $\gamma(\Phi;\Psi_{1},\dots,\Psi_{n})$ is an $m$-copula, where $\displaystyle m=\sum m_{k}$.
\end{proposition}

\begin{proof}
Groundedness and uniform margins follow directly from the definitions. For $m$-increasingness, note that if $V^{(k)}\sim\mathrm{Uniform}(\I^{m_{k}})$ independently, then $U_{k}=\Psi_{k}(V^{(k)})$ has uniform margins, and $(U_{1},\dots,U_{n})\sim\Phi$ yields a joint distribution function equal to the composite, hence it is an $m$-copula.
\end{proof}

\begin{theorem}
\label{thm:operad}
$(\mathfrak{C},\gamma,\mathrm{id},\cdot)$ is a symmetric operad in the category of sets.
\end{theorem}

\begin{proof}
Unit laws and associativity are immediate from function evaluation. Equivariance follows by renaming variables: for $\sigma\in S_{n}$,
\[
[\gamma(\Phi;\Psi_{1},\dots,\Psi_{n})]\cdot\tau_{\sigma}=\gamma(\Phi\cdot\sigma;\Psi_{\sigma^{-1}(1)},\dots,\Psi_{\sigma^{-1}(n)}),
\]
and for $\pi_{k}\in S_{m_{k}}$,
\[
\gamma(\Phi;\Psi_{1}\cdot\pi_{1},\dots,\Psi_{n}\cdot\pi_{n})=\gamma(\Phi;\Psi_{1},\dots,\Psi_{n})\cdot(\pi_{1},\dots,\pi_{n}),
\]
where $\tau_{\sigma}$ permutes blocks and $(\pi_{1},\dots,\pi_{n})$ permutes within blocks.
\end{proof}

Singular copulas (e.g., $M_{n}(u)=\min_{i}u_{i}$, $W_{2}(u,v)=\max(0,u+v-1)$) are elements of $\mathfrak{C}$; composition remains well-defined without requiring density functions.

\section{Absolutely continuous suboperad}
\label{sec:ac}
Let $\mathcal{C}_{n}^{ac}\subset\mathcal{C}_{n}$ be those copulas with absolutely continuous distribution and density $\displaystyle c=\frac{\partial^{n}C}{\partial u_{1}\cdots\partial u_{n}}\geq 0$, with $\displaystyle \int_{\I^{n}}c=1$.

\begin{proposition}
\label{prop:ac}
$\mathfrak{C}^{ac}=\bigsqcup_{n}\mathcal{C}_{n}^{ac}$ is a suboperad. For $\Phi\in\mathcal{C}_{n}^{ac}$, $\Psi_{k}\in\mathcal{C}_{m_{k}}^{ac}$, the composite has density
\[
g(v)=\phi\bigl(\Psi_{1}(v^{(1)}),\dots,\Psi_{n}(v^{(n)})\bigr)\prod_{k=1}^{n}\psi_{k}(v^{(k)}).
\]
\end{proposition}

\begin{proof}
This is the multivariate chain rule for Sklar's construction: each substitution $u_{k}=\Psi_{k}(v^{(k)})$ contributes the Jacobian factor $\psi_{k}$.
\end{proof}

\section{Copula entropy and additivity}
\label{sec:entropy}
For $C\in\mathcal{C}_{n}^{ac}$ with density $c$, define copula entropy following Ma and Sun \cite{masun}:
\[ \displaystyle H(C)=-\int_{\I^{n}}c(u)\log c(u)\,du. \]
Finite entropy requires $c\in L\log L(\I^{n})$, i.e., $\int_{\I^{n}}|c\log c|\,du<\infty$ \cite{rao-ren,bennett-sharpley}.

\begin{theorem}[Additivity]
\label{thm:add}
Let $\Phi\in\mathcal{C}_{n}^{ac}$, $\Psi_{k}\in\mathcal{C}_{m_{k}}^{ac}$ with densities $\phi,\psi_{k}$. Assume $\phi\in L\log L(\I^{n})$, $\psi_{k}\in L\log L(\I^{m_{k}})$, and the composite density $g\in L\log L(\I^{m})$. Then
\[ \displaystyle H\bigl(\gamma(\Phi;\Psi_{1},\dots,\Psi_{n})\bigr)=H(\Phi)+\sum_{k=1}^{n}H(\Psi_{k}). \]
\end{theorem}

\begin{proof}
Write $u_{k}=\Psi_{k}(v^{(k)})$. Then $\displaystyle \log g(v)=\log\phi(u)+\sum_{k=1}^{n}\log\psi_{k}(v^{(k)})$. The $L\log L$ hypotheses guarantee that $\log\phi$ and $\log\psi_{k}$ are integrable against the respective copula measures. Using the copula identity
\[ \displaystyle \int_{\I^{m_{k}}}h(\Psi_{k}(v^{(k)}))\psi_{k}(v^{(k)})\,dv^{(k)}=\int_{0}^{1}h(t)\,dt \]
for any integrable $h$, we compute:
\[ \displaystyle \int_{\I^{m}}g\log\phi\,dv=\int_{\I^{n}}\phi\log\phi\,du=-H(\Phi), \]
and for each $k$,
\[ \displaystyle \int_{\I^{m}}g\log\psi_{k}\,dv=\int_{\I^{m_{k}}}\psi_{k}\log\psi_{k}\,dv^{(k)}=-H(\Psi_{k}). \]
Summing gives the result.
\end{proof}

\section{Closed suboperads and counterexample}
\label{sec:closed}
To ensure $g\in L\log L$, we consider subclasses closed under composition.

\begin{proposition}
\label{prop:closed}
Each of the following families is a suboperad of $\mathfrak{C}^{ac}$ on which entropy is finite and additive:
\begin{enumerate}
\item[(1)] Bounded densities: $c\leq M$ for some $M>0$.
\item[(2)] $L^{p}$ densities for fixed $p>1$ (since $L^{p}\subset L\log L$ on a finite measure space).
\item[(3)] Boundary-growth densities: $c(u)\leq K\prod_{i=1}^{n}u_{i}^{-\alpha_{i}}(1-u_{i})^{-\beta_{i}}$ with $\displaystyle \sum_{i}(\alpha_{i}+\beta_{i})<1$.
\end{enumerate}
\end{proposition}

\begin{proof}
Boundedness and $L^{p}$ are preserved under products and substitutions. For boundary growth, consider the composite density $g(v)=\phi(\Psi_{1}(v^{(1)}),\dots,\Psi_{n}(v^{(n)}))\prod_{k}\psi_{k}(v^{(k)})$. Suppose each $\psi_{k}$ satisfies $\psi_{k}(v^{(k)})\leq K_{k}\prod_{i=1}^{m_{k}}(v_{i}^{(k)})^{-\alpha_{ki}}(1-v_{i}^{(k)})^{-\beta_{ki}}$ with $\displaystyle \sum_{i}(\alpha_{ki}+\beta_{ki})<1$, and $\phi(u)\leq K\prod_{j=1}^{n}u_{j}^{-\gamma_{j}}(1-u_{j})^{-\delta_{j}}$ with $\displaystyle \sum_{j}(\gamma_{j}+\delta_{j})<1$. For simplicity, assume each $\Psi_{k}$ is the independent copula $\Psi_{k}(v^{(k)})=\prod_{i=1}^{m_{k}}v_{i}^{(k)}$ near the origin; then $u_{k}=\prod_{i}v_{i}^{(k)}$, and the exponent of each $v_{i}^{(k)}$ in $g$ becomes $\alpha_{ki}+\gamma_{k}$. Summing over all variables gives $\displaystyle \sum_{k=1}^{n}\gamma_{k}+\sum_{k,i}(\alpha_{ki}+\beta_{ki})+\sum_{k}\delta_{k}=\sum_{k}(\gamma_{k}+\delta_{k})+\sum_{k,i}(\alpha_{ki}+\beta_{ki})<1$, maintaining the strict inequality.
\end{proof}

\begin{theorem}[Counterexample]
\label{thm:counterex}
There exist $\phi\in L\log L(\I^{2})$ and $\psi_{1},\psi_{2}\in L\log L(\I^{2})$ such that the binary composite has density outside $L\log L$ (hence entropy $-\infty$).
\end{theorem}

\begin{proof}
Consider Clayton copulas with parameters $\theta_{1},\theta_{2},\theta_{0}>0$. The density of a Clayton copula $C_{\theta}$ is
\[
c_{\theta}(u,v)=(1+\theta)(u^{-\theta}+v^{-\theta}-1)^{-1/\theta-2}u^{-\theta-1}v^{-\theta-1}.
\]
Near $(0,0)$, $\displaystyle c_{\theta}(u,v)\sim (1+\theta)(uv)^{-\frac{2\theta+1}{\theta+1}}$. Set $\psi_{k}=c_{\theta_{k}}$ with $\theta_{k}$ chosen so that $\displaystyle \frac{2\theta+1}{\theta+1}<1$ (i.e., $\theta_{k}<0$, but Clayton requires $\theta>0$ for this form). Instead, use the modified Clayton-type densities $\psi_{k}(x,y)=A_{k}(xy)^{-a_{k}}$ with $a_{k}<1$, normalized so that $\displaystyle \int_{0}^{1}\int_{0}^{1}\psi_{k}=1$. Similarly let $\phi(x,y)=B(xy)^{-b}$ with $b<1$. The composite on four variables behaves as $(u_{1}v_{1})^{-(a_{1}+b)}(u_{2}v_{2})^{-(a_{2}+b)}$ near zero. Choosing $a_{1}+b\geq 1$ makes $\displaystyle \int g|\log g|=\infty$, while each factor remains in $L\log L$ because $a_{k},b<1$.
\end{proof}

\section{Character interpretation}
\label{sec:character}
Let $\mathfrak{C}_{\mathrm{fin}}^{ac}$ be any closed suboperad from Section \ref{sec:closed}. Then $H:\mathfrak{C}_{\mathrm{fin}}^{ac}\to(-\infty,0]$ is an additive character:
\[ \displaystyle H(\gamma(\Phi;\Psi_{1},\dots,\Psi_{n}))=H(\Phi)+\sum_{k=1}^{n}H(\Psi_{k}). \]

\begin{example}
\label{ex:independent}
For the independent suboperad $\Pi_{n}(u)=\prod_{i=1}^{n}u_{i}$, the density is $c\equiv 1$, so $\displaystyle H(\Pi_{n})=-\int_{\I^{n}}1\log 1\,du=0$, consistent with vanishing mutual information.
\end{example}

Strong dependence (approaching $M_{n}$) drives $H\to-\infty$, consistent with mutual information $I\to+\infty$ via Ma--Sun's identity $I=-H_{c}$ \cite{masun}.
