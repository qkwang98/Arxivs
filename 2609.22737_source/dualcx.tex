
\begin{tikzpicture}[scale=6,
    % Arrow styles
    single/.style={postaction={decorate},decoration={markings, mark=at position 0.5 with {\arrow[scale=1.5]{>}}}},
    double/.style={postaction={decorate},decoration={markings, 
        mark=at position 0.48 with {\arrow[scale=1.5]{>}},
        mark=at position 0.52 with {\arrow[scale=1.5]{>}}}},
    triple/.style={postaction={decorate},decoration={markings, 
        mark=at position 0.47 with {\arrow[scale=1.5]{>}},
        mark=at position 0.5  with {\arrow[scale=1.5]{>}},
        mark=at position 0.53 with {\arrow[scale=1.5]{>}}}}
]

% Big triangle vertices
\coordinate (A) at (0,0);
\coordinate (B) at (1,0);
\coordinate (C) at (0.5,{sqrt(3)/2});

% Midpoints
\coordinate (D) at ($(A)!0.5!(B)$);
\coordinate (E) at ($(B)!0.5!(C)$);
\coordinate (F) at ($(C)!0.5!(A)$);

% Center of triangle DEF (centroid)
\coordinate (G) at (barycentric cs:D=1,E=1,F=1);

% Draw triangles
\draw (A) -- (B) -- (C) -- cycle;
\draw (D) -- (E) -- (F) -- cycle;
\draw (G) -- (D);
\draw (G) -- (E);
\draw (G) -- (F);


% Labels in the outer three small triangles
\node at (barycentric cs:A=1,D=1,F=1) {$v_{\gamma,1}'$};
\node at (barycentric cs:B=1,D=1,E=1) {$v_{\gamma,2}'$};
\node at (barycentric cs:C=1,E=1,F=1) {$v_{\gamma,3}'$};

% Labels in the three central sub-triangles (smaller font, shifted)
\node[xshift=0.05cm] at (barycentric cs:G=1,D=1,F=1) {$v_{\alpha,1}$}; % shifted right
\node[xshift=-0.05cm] at (barycentric cs:G=1,D=1,E=1) {$v_{\alpha,2}$}; % shifted left
\node[yshift=-0.04cm] at (barycentric cs:G=1,E=1,F=1) {$v_{\alpha,3}$}; % no shift


% Arrows
\draw[single] (F) -- (A);
\draw[single] (F) -- (C);

\draw[double] (D) -- (A);
\draw[double] (D) -- (B);

\draw[triple] (E) -- (B);
\draw[triple] (E) -- (C);


\end{tikzpicture}
