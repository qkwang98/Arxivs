\section{Tensor coinvariants, Schur resolutions and finite jets}
\label{sec:tensor-applications}

The selected-label presentation retains the algebra of finite jets.
A canonical Schur cover then describes every label and all its moment
homology. Retain the augmented
algebra $B$, its augmentation ideal $\mathfrak m$, and $Q_n(B)$ from
Section~\ref{sec:selected-quotients}; initially, only invertibility of
$n!$ in $K$ is required.

\begin{theorem}[First parabolics are jets]
\label{thm:universal-first-parabolic-jet}
For every augmented commutative algebra $B$ as above, inclusion of
the first tensor factor gives a natural algebra isomorphism
\begin{equation}\label{eq:universal-first-parabolic-jet}
                  Q_n(B)^{S_{n-1}}\cong B/\mathfrak m^n.
\end{equation}
For $n\ge2$, let $V_n=S^{(n-1,1)}$ be the standard representation.
There is a natural isomorphism
\begin{equation}\label{eq:universal-standard-jet}
 \operatorname{Hom}_{S_n}(V_n,Q_n(B))
       \cong \mathfrak m/\mathfrak m^n
\end{equation}
of modules for every augmentation-preserving symmetry group of $B$.
If $B$ is graded and $\varepsilon$ preserves the grading, both
isomorphisms preserve the grading.
\end{theorem}

\begin{proof}
Take $r=1$ in Theorem~\ref{thm:universal-selected-trace-presentation}.
The relation is $b^n=0$ for every $b\in\mathfrak m$.
The ideal generated by these powers is exactly $\mathfrak m^n$:
the reverse inclusion follows by polarizing the $n$th power, or
directly from
\[
 n!\,b_1\cdots b_n
  =\sum_{J\subseteq\{1,\ldots,n\}}(-1)^{n-|J|}
       \left(\sum_{j\in J}b_j\right)^n.
\]
Moreover $Q_n(B)^{S_n}=K$, by Reynolds averaging and the definition
of the invariant-generated ideal.  The permutation module
$\operatorname{Ind}_{S_{n-1}}^{S_n}\mathbf1$ splits as
$\mathbf1\oplus V_n$.  Frobenius reciprocity therefore identifies
the standard multiplicity space with the augmentation kernel of
$Q_n(B)^{S_{n-1}}$, which is $\mathfrak m/\mathfrak m^n$ under
\eqref{eq:universal-first-parabolic-jet}.
Concretely, the reciprocity map evaluates a homomorphism on
$e_1-n^{-1}\sum_i e_i$, so the identification is natural.
\end{proof}

For $B=\Sym V$, the standard multiplicity space is
$\bigoplus_{d=1}^{n-1}\Sym^dV$. This polynomial specialization
recovers the bosonic standard-character formula of
Jiang--Lentfer~\cite[Proposition 6.5 and Theorem 6.6]{JiangLentfer2026};
their theorem also treats fermionic alphabets. The statement above
identifies the algebra and its augmentation kernel for arbitrary
augmented $B$.

\subsection{Natural stable multiplicity spaces}
\label{sec:stable-specht-multiplicities}

Let $B=K\oplus\mathfrak m$ be a connected positively graded
commutative algebra over a field of characteristic zero. Retain $Q_n(B)$ from
Theorem~\ref{thm:universal-selected-trace-presentation}.
For $\tau\vdash s$ write $\tau[n]=(n-s,\tau)$ when this is a partition.
For the empty partition set $\tau_1=0$.

\Needspace{13\baselineskip}
\begin{samepage}
\begin{theorem}[Natural stable multiplicity spaces]
\label{thm:stable-augmented-multiplicities}
If $D\ge s$ and $n\ge D+\tau_1$, then
\[
 \operatorname{Hom}_{S_n}(S^{\tau[n]},Q_n(B)_D)
 \simeq (\mathbb S_\tau\mathfrak m)_D.
\]
Fixing the one-dimensional Pieri branching line makes the
isomorphism natural for graded augmented algebra maps and
auxiliary symmetries preserving the grading and augmentation.
For $D<s$ the left side is zero whenever $\tau[n]$ is a partition.
\end{theorem}
\end{samepage}
\begin{proof}
The canonical cover in Theorem~\ref{thm:two-schur-presentation} has
no relation in degrees below $n+1-\tau_1$, by
Proposition~\ref{prop:first-schur-obstruction}. Its quotient modulo
positive traces is therefore exactly $\Schur_\tau\mathfrak m$ in
the stated range. This is the actual minimum-occupancy quotient map,
so it has the asserted naturality. For $D<s$, occupancy gives zero.
\end{proof}

\begin{corollary}[Universal separation and its sharp range]
\label{cor:universal-augmented-separation}
For $D\ge1$ and $n\ge2D-1$, multiplication gives a canonical isomorphism
\[
 \bigl(P_n(B)^{S_n}\otimes L_n(B)\bigr)_{\le D}
                     \xrightarrow{\sim}P_n(B)_{\le D},
\]
and all positive moment homology vanishes through degree $D$.
The bound is sharp for arbitrary connected graded augmented algebras.
\end{corollary}
\begin{proof}
A degree-$d$ label has tail size at most $d$. At $n=2d-1$, a tail
with first part $d$ would have to be $(d)$ and would give the invalid
label $(d-1,d)$. Thus every occurring label satisfies the individual
bound $n\ge d+\tau_1$; larger $n$ are immediate. The trivial-label
presentation shows that $S_B$ and the actual invariant algebra agree
through degree $n$, giving the multiplication statement.

For sharpness let $D\ge2$, $n=2D-2=2m$, and $B=K[x]/(x^2)$.
The sole moment $p=\sum_{i=1}^{2m}x_i$ kills the nonzero element
$f=\prod_{j=1}^m(x_{2j-1}-x_{2j})$. Hence $f e_{T_x}$ is nonzero
first Koszul homology in degree $D$; a one-generator complex has no
boundaries there. The endpoint $D=n=1$ is included in the isomorphism.
\end{proof}

For a fixed tail, augmentation of the last factor gives compatible
surjections $M_{\tau[n]}(B)\to M_{\tau[n-1]}(B)$ whenever both labels
are defined. With compatible Pieri lines they are quotients of the
identity on $\Schur_\tau\mathfrak m$, so their kernels decrease with
$n$. Indeed, the occupancy-$s$ projection is onto, and among its source
labels of first row at least $n-s$, only $\tau[n]$ branches to
$\tau[n-1]$. Multiplicity monotonicity itself is known in the FI-module
setting \cite[Lemma~2.1.1]{Borisov2026}; here the common source and maps
are explicit for arbitrary augmented algebras.



\subsection{Polynomial algebras and positive plethysm}

Write
\[
 A=\operatorname{Sym}V,
 \qquad A_+=\bigoplus_{d\ge1}\operatorname{Sym}^dV,
 \qquad H(\mathbf q)=\prod_{a=1}^k(1-q_a)^{-1}.
\]
The grading on the Schur functor $\mathbb S_\tau(A_+)$
is induced by the usual polynomial grading of $A_+$.
In particular, every homogeneous part is finite dimensional.

\begin{corollary}[Explicit stable Specht multiplicities]
\label{cor:explicit-stable-specht-multiplicities}
For $D\ge1$ and $n\ge2D-1$, there is an isomorphism of
$\operatorname{GL}(V)\times S_n$-modules
\begin{equation}\label{eq:stable-specht-schur-functor}
 (R_{n,k})_D\cong
 \bigoplus_{\substack{|\tau|\le D\\\tau[n]\ \mathrm{a\ partition}}}
   \bigl(\mathbb S_\tau(A_+)\bigr)_D
       \otimes S^{(n-|\tau|,\tau)}.
\end{equation}
Thus the multigraded stable multiplicity of the label
$\tau[n]=(n-|\tau|,\tau)$ is the positive plethysm
\begin{equation}\label{eq:stable-specht-plethysm}
 s_\tau[H(\mathbf q)-1].
\end{equation}
For a fixed nonempty $\tau$ and
$D\ge|\tau|$, its degree-$D$ multiplicity is
already given by this formula when
\begin{equation}\label{eq:individual-stable-specht-range}
 n\ge D+\tau_1.
\end{equation}
The term for $\tau=\varnothing$ is the trivial label,
which occurs only in degree zero.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:stable-augmented-multiplicities} to
$B=\operatorname{Sym}V$. Corollary~\ref{cor:universal-augmented-separation} covers
every valid label when $n\ge2D-1$. The same theorem gives the sharper individual
range for every $D\ge1$. Schur functor characters give
\eqref{eq:stable-specht-plethysm}, in each multidegree.
\end{proof}

For example, the tails $\tau=(1),(2),(1,1)$ give
$A_+$, $\operatorname{Sym}^2(A_+)$ and $\bigwedge^2(A_+)$,
respectively, before expansion in row Schur functions.
