\section{Mapping objects and comparison theorems}
\label{sec:mapping}\label{sec:comparison}\label{sec:base}

The categories $\M_n(\mathcal A)$ in this section are those constructed
by localization in Section~\ref{sec:presheaf}. We now compute their
mapping objects. In particular, the formulas below are consequences
of that construction, rather than definitions of the motives.
Ordinary $K$ denotes connective algebraic K-theory of a small
idempotent-complete stable category.

\begin{theorem}\label{thm:mapping-comparison}
Let $n\geq1$ and let $\mathcal A\in\nPr{(n+1)}$.
For $c\in\mathcal A^\omega$ and arbitrary
$a\in\mathcal A$, there are natural equivalences
\begin{align}
 \operatorname{map}_{\M_1(\mathcal A)}(U_{\mathcal A}c,U_{\mathcal A}a)
  &\simeq K(\PrHom_{\mathcal A}(c,a)^\omega)
       \qquad(n=1),\label{eq:mapping-height-one}\\
 \PrHom_{\M_n(\mathcal A)}(U_{\mathcal A}c,U_{\mathcal A}a)
  &\simeq\M_{n-1}(\PrHom_{\mathcal A}(c,a))
       \qquad(n\geq2).\label{eq:mapping-higher}
\end{align}
The first line uses the compact stable category of the coefficient
Hom. In the second line the argument is the presentable coefficient
Hom itself, an object of $\nPr n$.

These equivalences preserve composition and units. They are natural
for compact-preserving module functors and after every coefficient
probe. They also intertwine the external products as lax symmetric
monoidal constructions.
\end{theorem}

We prove the first coefficient level directly. The remaining levels
follow from the already established additive universal properties.






























\subsection{The first coefficient level}

Fix $\mathcal A\in\nPr2$ and put $C=\mathcal A^\omega$.
Let $\mathcal X$ be the small stable $2$-category
with objects $C$ and Hom categories
\[
 \mathcal H(c,d)=\PrHom_{\mathcal A}(c,d)^\omega.
\]
Its composition is biexact, and its finite
biproducts are inherited from $\mathcal A$.
The complete exact sequences of
Definition~\ref{def:genuine} agree with the
$1$-localization sequences of
\citep[Definition~1.2.2]{MGS}.
Consequently, the two spectral-presheaf
localizations agree, giving
\[
 \M_1(\mathcal A)
 \simeq\operatorname{Mot}^{\mathrm{MGS}}(\mathcal X).
\]

Write
\[
 \mathscr P_C=\Fun(C^{\op},\Sp),
 \qquad
 y_b=\Sigma^\infty_+\Map_C(-,b),
\]
and let
$L:\mathscr P_C\to\M_1(\mathcal A)$
be the localization.
For $b\in C$, put
\[
 K_b(c)=K(\mathcal H(c,b)).
\]

The proof of \citep[Theorem~2.3.1]{MGS}
shows that $K_b$ is $L$-local and that the
canonical object-to-$K$ map induces an
equivalence
\begin{equation}\label{eq:height-one-unit}
 Ly_b\xrightarrow{\sim}K_b.
\end{equation}
We now establish the coherent multiplicative
form of this comparison.

For a small spectral category $\mathbb B$, write
\[
 \Mod(\mathbb B)=\Pmod_{\Sp}(\mathbb B),
 \qquad
 \Perf(\mathbb B)=\Mod(\mathbb B)^\omega.
\]

\begin{proposition}\label{prop:height-one-presentation}
Let $\mathbb K_C$ be the spectral category
with objects $C$ and mapping spectra
\[
 \mathbb K_C(c,d)=K(\mathcal H(c,d)).
\]
Composition and units are induced by the
lax symmetric monoidal structure of $K$
\citep{BGTMult}.
There is a canonical equivalence
\begin{equation}\label{eq:height-one-presentation}
 \M_1(\mathcal A)\simeq\Mod(\mathbb K_C)
\end{equation}
sending $U_{\mathcal A}b$, for compact $b$,
to the represented module at $b$.
For arbitrary $a\in\mathcal A$, its image
is the module
\[
 c\longmapsto
 K(\PrHom_{\mathcal A}(c,a)^\omega).
\]
The equivalence is natural in $\mathcal A$
and compatible with composition, units,
and external products, including their
coherent finite-arity operations.
\end{proposition}


\begin{proof}
The multiplicative object-to-$K$ map
\citep{BGTMult} induces a spectral functor
\[
 \Sigma^\infty_+C\longrightarrow\mathbb K_C.
\]
Restriction of scalars sends the represented
module $h_b$ to $K_b$. 
By \eqref{eq:height-one-unit}, these
presheaves are local and satisfy
$K_b\simeq Ly_b$.
Since local presheaves
are closed under colimits, restriction
factors through
\[
 R:\Mod(\mathbb K_C)\longrightarrow\M_1(\mathcal A).
\]
On represented modules, the induced map
on mapping spectra is the identity
\[
 \mathbb K_C(c,b)=K_b(c).
\]
Thus $R$ is an equivalence by compact
generation and spectral Yoneda.

For arbitrary $a\in\mathcal A$, the asserted
formula follows by filtered continuity of
$\PrHom_{\mathcal A}(c,-)$ and $K$.
Finally, compatibility with external products
follows from the multiplicativity of the
object-to-$K$ map and colimit preservation,
including the coherent finite-arity operations.
\end{proof}

















































\subsection{Proof of Theorem~\ref{thm:mapping-comparison}}


\begin{proof}[{Proof of Theorem~\ref{thm:mapping-comparison}}]
We argue by induction on $n$, the case $n=1$
having already been established.
For $n\geq2$, let
\[
 \mathcal D_A^{(n)}(a,b)
 =\M_{n-1}(\PrHom_A(a,b))
\]
and put
\[
 \M'_n(A)=
 \Pmod_{\nPr{(n-1)}}(\mathcal D_A^{(n)}).
\]
Composition and units are induced by the external
products at height $n-1$.
We claim that \(\M_n(A)\simeq\M'_n(A)\).

The natural functor
$A^\omega\to\M'_n(A)$ is additive.
Indeed, applying $\PrHom_A(t,-)$ to an exact
sequence and using
Lemma~\ref{lem:module-row-preservation}
shows that its image is exact after evaluation
at every representable.
The universal property of $\M_n(A)$ therefore
gives a coefficient-linear functor
\[
 \Phi:\M_n(A)\longrightarrow\M'_n(A).
\]

Conversely, the local diagrams of $U_A$ are additive.
Indeed, a compact local exact sequence is classified
by a map $W_n\odot a\to b$, and the universal upper
exact sequence \eqref{eq:presentable-upper-row}
shows that its image under $U_A$ is exact.
By the inductive universal property, these
local diagrams extend to compact-preserving
coefficient maps
\[
 \M_{n-1}(\PrHom_A(a,b))
 \longrightarrow
 \PrHom_{\M_n(A)}(U_Aa,U_Ab).
\]
Their compatibility with composition and units
follows from the parameterized universal property
at height $n-1$. Enriched Yoneda gives
\[
 \Psi:\M'_n(A)\longrightarrow\M_n(A).
\]

Since \(\Psi\Phi U_A=U_A\), \(\Psi\Phi\) is
equivalent to the identity by the universal
property of $\M_n(A)$.
Similarly, $\Phi\Psi$ is the identity on
representables and on their coefficient Homs:
the latter follows by the inductive universal
property from the identity on represented
local generators.
Enriched Yoneda therefore gives
$\Phi\Psi\simeq\mathrm{id}$.


Enriched Yoneda gives the asserted mapping formula
when both objects are compact.
For an arbitrary target
$a=\operatorname*{colim}_i a_i$ with compact $a_i$,
both sides preserve this filtered colimit.
On the left, this follows from the compactness
of $U_Ac$ and the filtered continuity of $U_A$.
On the right, the same follows from
compact-source continuity of
$\PrHom_A(c,-)$ and
Lemma~\ref{lem:motive-filtered-colimits}.
This proves \eqref{eq:mapping-higher}.



Finally, we verify compatibility with external products.
For any finite family $(A_i)_{i=1}^m$, the external
product on the coefficient categories
$\mathcal D_{A_i}^{(n)}$ is induced by
\[
\begin{aligned}
 \bigotimes_i \M_{n-1}(\PrHom_{A_i}(a_i,b_i))
 &\longrightarrow
 \M_{n-1}\Bigl(
   \bigotimes_i\PrHom_{A_i}(a_i,b_i)\Bigr)\\
 &\longrightarrow
 \M_{n-1}\Bigl(
   \PrHom_{\bigotimes_i A_i}
   (\bigotimes_i a_i,\bigotimes_i b_i)\Bigr).
\end{aligned}
\]
By Proposition~\ref{prop:motivic-products},
these maps are coherent in all arities and induce
external products on the enriched presheaf categories
$\M'_n(A)$.

The comparison functors $\Phi$ and $\Psi$ intertwine
these products on representables and their coefficient
Homs, by the inductive universal property.
The parameterized form of that property identifies
all finite-arity operations and their coherences.
Since representables and their coefficient actions
generate under colimits, the compatibility extends
to all objects. Thus the equivalence
$\M_n(A)\simeq\M'_n(A)$ is compatible with
external products as a lax symmetric monoidal
equivalence.

\end{proof}



\section{Higher K-theory}\label{sec:higher-k}


The regular input $\one_{n+1}$ has underlying category $\nPr n$.
Its distinguished compact object is $e_n=\one_n$.

\begin{definition}\label{def:unit-corepresented-K}
Write
$\operatorname{Mot}_n=\M_n(\one_{n+1})$ and
$\xi_n=U_{\one_{n+1}}(e_n)$.
Day convolution has unit $\xi_n$.
For $A\in\nPr n$ define
\begin{equation}
 K^{[n]}(A)=
 \begin{cases}
  \operatorname{map}_{\operatorname{Mot}_1}(\xi_1,U(A)),&n=1,\\
  \PrHom_{\operatorname{Mot}_n}(\xi_n,U(A)),&n\geq2.
 \end{cases}
 \label{eq:unit-corepresented-K}
\end{equation}
\end{definition}


By Proposition~\ref{prop:motivic-products},
the functor $K^{[n]}$ is lax symmetric monoidal.
In particular, for any $\mathbb E_k$-algebra
$A\in\nPr n$, $k\geq1$, the object
$K^{[n]}(A)\in\nPr{(n-1)}$ naturally inherits
an $\mathbb E_k$-algebra structure.


\begin{corollary}\label{cor:K-normalization}
Naturally in $A\in\nPr n$, there are equivalences
\begin{equation}\label{eq:K-normalization}
 K^{[1]}(A)\simeq K(A^\omega)\quad(n=1),
 \qquad
 K^{[n]}(A)\simeq\M_{n-1}(A)\quad(n\geq2).
\end{equation}
These equivalences are lax symmetric monoidal.
In particular, if $A$ is an $\mathbb E_k$-algebra
for $k\geq1$, the equivalences are equivalences of
$\mathbb E_k$-algebras.
\end{corollary}

\begin{proof}
The coefficient Hom from $e_n$ in the regular
module is $A$ itself. Applying
Theorem~\ref{thm:mapping-comparison} gives
the equivalences. Their compatibility with
composition, units, and external products
identifies the lax symmetric monoidal
structures, hence also the induced
$\mathbb E_k$-algebra structures.
\end{proof}



\begin{corollary}\label{cor:compact-internal-hom}
Let $A$ be compact as an object of $\nPr n$ and let $B\in\nPr n$
be arbitrary. Denote by $\operatorname{Func}(A,B)$ the right adjoint
to the tensor product in this specified compact-preserving
coefficient category. Then
\begin{equation}\label{eq:compact-internal-hom}
 \PrHom_{\operatorname{Mot}_n}(U(A),U(B))
 \simeq
 \PrHom_{\operatorname{Mot}_n}
       \bigl(\xi_n,U(\operatorname{Func}(A,B))\bigr).
\end{equation}
For $n=1$ both sides are mapping spectra.
The equivalence is contravariantly natural in compact $A$ and
covariantly natural in $B$.
\end{corollary}

\begin{proof}
In the regular module, tensor--Hom adjunction gives
\[
 \PrHom_{\one_{n+1}}(A,B)\simeq\operatorname{Func}(A,B),
 \qquad
 \operatorname{Func}(e_n,H)\simeq H.
\]
Apply Theorem~\ref{thm:mapping-comparison} first to the compact source
$A$ and then to the compact source $e_n$, with target
$H=\operatorname{Func}(A,B)$.
Both sides of \eqref{eq:compact-internal-hom} identify with
$K^{[n]}(H)$.
\end{proof}

Compactness here is a condition on $A$ in the ambient coefficient
category, not merely compact generation of its underlying category.
For $n=1$ the internal Hom is
$\Ind\Fun^{\ex}(A^\omega,B^\omega)$.
It is not the category of all colimit-preserving functors
$A\to B$.

\begin{corollary}\label{cor:nonzero-all-heights}
For every finite $n\geq1$, the motive $\M_n(\one_{n+1})$ is nonzero.
More precisely,
\[
 \operatorname{map}_{\M_1(\one_2)}(\xi_1,\xi_1)\simeq K(\mathbb S),
 \qquad
 \PrHom_{\M_n(\one_{n+1})}(\xi_n,\xi_n)
       \simeq\M_{n-1}(\one_n)\quad(n\geq2).
\]
\end{corollary}



\begin{corollary}\label{cor:general-MGS}
Let $\mathcal A\in\nPr2$, and let
$\mathcal C_{\mathcal A}$ be the small stable
$2$-category with objects $\mathcal A^\omega$
and Hom categories
\[
 \operatorname{hom}_{\mathcal C_{\mathcal A}}(c,d)
 =\PrHom_{\mathcal A}(c,d)^\omega.
\]
Then there are natural equivalences
\[
 \begin{aligned}
 \operatorname{Mot}^{\mathrm{MGS}}
       (\mathcal C_{\mathcal A})
   &\simeq  K^{[2]}(\mathcal A)
     ,\\
 K^{(2,1)}_{\mathrm{MGS}}
       (\mathcal C_{\mathcal A})
   &\simeq (K^{[2]}(\mathcal A))^\omega,\\
 K^{(2)}_{\mathrm{MGS}}
       (\mathcal C_{\mathcal A})
   &\simeq K((K^{[2]}(\mathcal A))^\omega).
 \end{aligned}
\]
See \cite{MGS} for the notations.
\end{corollary}

\begin{proof}
The first equivalence follows from
Corollary~\ref{cor:K-normalization}
and the comparison with MGS in
Section~\ref{sec:mapping}.
The remaining equivalences follow
from the definitions in \citep{MGS}.
\end{proof}
