\section{Universal exact sequences}
\label{sec:compact-classifiers}\label{sec:weights}

We construct a module classifying exact sequences
directly in the coefficient category.
Ordinary compact objects, duality, and Ind-completion
suffice for its universal property.

\subsection{Coefficient limits and diagram modules}


\begin{lemma}
\label{lem:compact-coefficient-homs}
\label{lem:compact-hom-filtered}
Let $s\geq1$. The forgetful functor
$\nPr s\to\PrL$ creates limits and filtered colimits.
Taking compact objects preserves these operations,
computed in the category of small idempotent-complete
infinity-categories with finite colimits.

For $H=\lim_i H_i$ in $\nPr s$ and $a,b\in H^\omega$,
with images $a_i,b_i$, we have
\[
 \PrHom_H(a,b)
 \simeq\lim_i\PrHom_{H_i}(a_i,b_i).
\]
For a filtered colimit $H=\operatorname*{colim}_i H_i$
and compact $a,b\in H$, choose a common stage $i_0$
representing both objects. Then
\[
 \PrHom_H(a,b)
 \simeq
 \operatorname*{colim}_{(i_0\to j)\in I_{i_0/}}
 \PrHom_{H_j}(a_j,b_j).
\]
Both equivalences are in $\nPr{(s-1)}$.

Moreover, for $H\in\nPr s$ and $a\in H^\omega$,
the functor
$\PrHom_H(a,-):H\to\nPr{(s-1)}$
preserves filtered colimits.
\end{lemma}

\begin{proof}
The first assertion follows from the fixed-cardinal
module formalism \citep[Section~2.1]{Aoki}
and the ordinary Ind-equivalence \citep{LurieHTT}.

For compact $v\in\nPr{(s-1)}$, tensor--Hom
adjunction gives
\begin{equation}\label{eq:fnd-compact-hom-test}
 \Map_{\nPr{(s-1)}}(v,\PrHom_H(a,b))
 \simeq\Map_H(v\odot a,b).
\end{equation}
Since $v\odot a$ is compact, mapping spaces in
the compact cores commute with the indicated
limits and filtered colimits. Compact coefficients
detect equivalences, proving both Hom formulas.

Finally, for a filtered diagram $(b_j)$ in $H$,
compactness of $v\odot a$ gives
\[
 \Map_H(v\odot a,\operatorname*{colim}_j b_j)
 \simeq\operatorname*{colim}_j\Map_H(v\odot a,b_j).
\]
Applying the same tensor--Hom adjunction and
compact detection proves the last assertion.
\end{proof}








A small coefficient category $\mathcal D$ enriched in
$\nPr s$ is an associative algebra of object-indexed coefficient
matrices. Its coherent right modules form
\[
 \Pmod_{\nPr s}(\mathcal D)=\Fun_{\nPr s}(\mathcal D^{\op},\nPr s),
 \qquad h_d(c)=\mathcal D(c,d).
\]
The coefficients in this formula are the presentable objects of
$\nPr s$ themselves. Its parameter-compatible free cocompletion property
is Lemma~\ref{lem:parameter-modules}.



\begin{lemma}\label{lem:parameter-modules}


Let $s\geq1$, let $\mathcal D$ be a small category
enriched in $\nPr{(s-1)}$, and let $E\in\nPr s$.
The diagram module $E^{\mathcal D}$ belongs to
$\nPr s$, with pointwise colimits and coefficient
action. In particular,$\Pmod_{\nPr{(s-1)}}(\mathcal D)
 \simeq \one_s^{\mathcal D^{\op}}\in\nPr s.$
Restriction along the Yoneda embedding gives
\begin{equation}\label{eq:full-presheaf-up}
 \Fun^L_{\nPr{(s-1)}}(\Pmod_{\nPr{(s-1)}}(\mathcal D),E)
 \simeq\Fun_{\nPr{(s-1)}}(\mathcal D,E).
\end{equation}
This equivalence preserves all coefficient intertwining objects.
The represented objects are compact, but their images on the right
may be arbitrary objects of $E$.
\end{lemma}

\begin{proof}
Choose a set of objects $I$ and write $B_{xy}=\mathcal D(x,y)$.
The diagram module $E^{\mathcal D}$ is the category of coherent modules for
\[
 (TF)_y=\coprod_{x\in I}B_{xy}\odot F_x
 \quad\text{on }\prod_{x\in I}E.
\]
Its presentability, pointwise colimits, and free evaluations follow
from \cite[Theorem~4.43 and Corollary~4.44]{HeineBi}. The formulas
\[
 (L_xc)_y=B_{xy}\odot c,\qquad
 \Map(L_xc,F)\simeq\Map_E(c,F_x)
\]
show that compact $c$ gives compact $L_xc$. These objects generate,
and $v\odot L_xc\simeq L_x(v\odot c)$ is compact for compact $v$.
This proves the required compact generation and action condition.
No equality with pointwise compactness of arbitrary diagrams is used.
The coefficient Hom is
\begin{equation}\label{eq:parameter-bar-hom}
 \PrHom_{E^{\mathcal D}}(F,G)
 \simeq\operatorname*{Tot}_{[m]\in\Delta}
 \prod_{y\in I}\PrHom_E((T^mUF)_y,(UG)_y),
\end{equation}
in $\nPr{(s-1)}$, with operators induced by the actions
and the monad
\citep[Corollary~4.49]{HeineBi}. The free cocompletion theorem
\citep[Theorems~3.41(4) and~4.84]{HeineBi} gives
\eqref{eq:full-presheaf-up}, including these coefficient Homs.
\end{proof}


\begin{definition}\label{def:free-linearization}
For $s\geq1$ and a small category $K$, put
\[
 \mathsf F_s(K)=\Fun(K^{\op},\nPr{(s-1)})\in\nPr s.
\]
\end{definition}

Applying Lemma~\ref{lem:parameter-modules} to the free
$\nPr{(s-1)}$-enrichment of $K$ gives a natural equivalence
\begin{equation}\label{eq:free-linearization-full-up}
 \Fun^L_{\nPr{(s-1)}}(\mathsf F_s(K),E)
 \simeq\Fun(K,E),
 \qquad E\in\nPr s,
\end{equation}
including coefficient Homs.
In particular, $\mathsf F_s(*)\simeq\one_s$ and
$\mathsf F_s([1])$ represent compact objects and arrows
between compact objects, respectively. Both modules are
compact in $\nPr s$, since these classification functors
preserve filtered colimits by
Lemma~\ref{lem:compact-coefficient-homs}
and the finiteness of $[1]$.


\begin{remark}\label{rem:parameter-detection}
We detect coefficient Homs by all compact $v\in \nPr{(s-1)}$. Let $J(v)$
have two objects, endomorphism coefficients $\one_{s-1}$, coefficient
$v$ from the first to the second, and zero in the reverse direction.
An object of $E^{J(v)}$ is $(x_0,x_1,v\odot x_0\to x_1)$.
For these finite acyclic parameters, intrinsic compactness is
pointwise. Free evaluations on compact objects have compact values
and generate. Conversely, mapping out of a pointwise compact
diagram is a finite limit of mapping spaces out of compact objects,
using compactness of $v\odot x$, and preserves filtered colimits.
Finite chains detect ordinary functor categories, and these two-object
parameters detect their coefficient intertwining objects.
\end{remark}





\subsection{Exact sequences and their compact classifier}

\begin{lemma}\label{lem:matrix-adjunctions}
For $H\in\nPr s$, $s\geq2$, and an arrow $i:a\to b$ in $H$, coherent
right-adjoint data have an empty or contractible space. Existence
is detected by a reverse arrow, a unit, a counit, and the triangle
identities in the homotopy category of $\mathcal B_H(a,b)$ (see Definition \ref{def:adjunction}).
These spaces preserve limits and filtered colimits with compact
outer endpoints.
\end{lemma}


\begin{proof}
Let $I$ be the matrix in $\mathcal B_H(a,b)$ whose only
nonzero entry is $i$. A right adjoint $r$ of $i$
determines a right dual of $I$, given by the matrix
with $r$ in the opposite off-diagonal entry.
The corresponding unit, counit, and triangle identities
agree with those of the adjunction $i\dashv r$.
Conversely, the diagonal idempotents recover the
opposite off-diagonal entry of any right dual of $I$.
Thus the two constructions give equivalent spaces
of choices.

The existence of a right dual can be checked in the
homotopy category, and its coherent structure is
unique up to a contractible space of choices
\citep[Lemmas~4.6.1.6 and~4.6.1.10]{LurieHA}.
In particular, arbitrary choices of homotopies
witnessing the triangle identities need not themselves
form coherent duality data.

Dual pairs are preserved by limits of monoidal
categories \citep[Proposition~4.6.1.11]{LurieHA}.
For filtered colimits, Lemma~\ref{lem:compact-coefficient-homs}
allows the objects, morphisms, and triangle identities
to be represented at a common stage.
Uniqueness of right adjoints then gives the claimed
compatibility of the spaces of coherent adjunction data.
\end{proof}





\begin{lemma}\label{lem:exact-row-structure}\label{lem:fnd-parameters}

Let $s\geq1$, let $\mathcal D$ be a small category
enriched in $\nPr{(s-1)}$, and let $E\in\nPr s$.
An exact sequence in $E^{\mathcal D}$ consists of
pointwise exact sequences in $E$ whose adjunctions
assemble coherently, satisfying the Beck--Chevalley
conditions along the morphisms of $\mathcal D$.
Morphisms between such exact sequences are pointwise
morphisms compatible with the specified adjunctions,
including their units and counits, and satisfying the
Beck--Chevalley conditions.
Their coefficient Homs are the corresponding
ends. Exactness of a sequence is detected
at its vertices. The functor
\[
 \Phi_s^\omega(H)=\EqRow_s^\omega(H)^\simeq
       \colon\nPr s\longrightarrow\mathcal S
\]
preserves limits and filtered colimits.
\end{lemma}


\begin{proof}
At $s=1$, cofiber sequences are computed pointwise.
For $s\geq2$, pointwise adjunctions assemble precisely
when the Beck--Chevalley mates are invertible
\citep[Proposition~7.3.2.11]{LurieHA}.
The same criterion applies to enriched structure maps,
with coherence inherited from the mate correspondence.
The central exact sequences assemble by induction.
For a map of exact sequences with middle component $f_b:b\to b'$, compare
both central rows in $\PrHom_H(b,b')$ by whiskering. The strong
comparisons identify $f_bir$ with $i'r'f_b$ and $f_bjp$ with
$j'p'f_b$, together with the units and counits. Dual uniqueness
compares their lower adjunction data. 
The specified central zero comparison is canonical by adjunction
and $pi=0$. At the base its cofiber condition is a property of
that specified square. Induction proves the pointwise assertion
and its compatibility with coefficient ends.

In a limit, compact endpoints and their coefficient Homs are
computed by Lemma~\ref{lem:compact-coefficient-homs}. Adjunctions
glue by Lemma~\ref{lem:matrix-adjunctions}, and the central row
glues by the lower-stage assertion. For a filtered colimit,
the same lemmas descend the endpoints, adjunctions, invertible
comparisons, and recursively the central row to a stage. Finite
limits of spaces commute with filtered colimits. Dual uniqueness
and lower induction identify all paths and their compatibilities,
proving the space-valued assertion.
\end{proof}

\begin{proposition}\label{prop:fnd-classifier}\label{def:presentable-exact-classifier}
Let \(s\ge1\).
There is a compact module $W_s\in\nPr s$,
with a distinguished exact sequence of compact objects, such that
\begin{equation}\label{eq:fnd-space-classifier}
 \Map_{\nPr s}(W_s,H)\simeq\Phi_s^\omega(H).
\end{equation}
\end{proposition}

\begin{proof}
The preceding lemma makes $\Phi_s^\omega$ accessible and
limit preserving. The ordinary adjoint functor theorem
\citep[Section~5.5.2]{LurieHTT} gives a left adjoint $F_s$.
Take $W_s=F_s(*)$. Filtered-colimit preservation gives its
compactness. Its distinguished row corresponds to the identity.
\end{proof}

For example, at $s=1$ one may take $W_1=\Fun([1],\Sp)$, with universal exact sequence
at $(\mathbb S\xrightarrow1\mathbb S)$.

\begin{definition}
For \(s\ge 1\), we call \(W_s\) the universal exact-sequence module.
\end{definition}


\begin{corollary}\label{cor:presentable-exact-classifier}
For \(s\ge1\), there are natural equivalences
\begin{align}
 \Fun^L_{\nPr{(s-1)}}(W_s,H)
 &\simeq\EqRow_s(H),\label{eq:presentable-exact-classifier}\\
 \Fun^{L,\omega}_{\nPr{(s-1)}}(W_s,H)
 &\simeq\EqRow_s^\omega(H).\label{eq:compact-exact-classifier}
\end{align}
They preserve all coefficient intertwining objects. The first
equivalence commutes with coefficient diagram parameters.
\end{corollary}

\begin{proof}
First upgrade \eqref{eq:fnd-space-classifier} to the second
assertion. Apply \eqref{eq:fnd-space-classifier} to the finite
parameter modules of Remark~\ref{rem:parameter-detection}, where
compactness is pointwise, and use
Lemma~\ref{lem:fnd-parameters}. Finite chains identify the ordinary
functor categories. The $J(v)$ tests identify every coefficient Hom
by Remark~\ref{rem:parameter-detection}. This proves the complete
compact assertion, without an assumption about arbitrary parameters.


Let us prove the first assertion.
For any small set of objects of $H$, let $C\subset H$
be the full subcategory generated under finite colimits,
retracts, and compact coefficient actions.
Then $K_C=\Ind(C)\in\nPr s$, and the canonical
coefficient-linear left adjoint $K_C\to H$ is
fully faithful on coefficient Homs between objects of $C$.
Indeed, this follows from the tensor--Hom adjunction
and the full faithfulness of $C\subset H$.
Coefficient full faithfulness lifts its entire
recursive data to a compact exact sequence in $K_C$, giving a functor
$W_s\to K_C\to H$. Conversely, an arbitrary coefficient-linear
left adjoint $F:W_s\to H$ lifts through such a $K_C$ after including
$F(W_s^\omega)$ in $C$. Its restriction to $W_s^\omega$ preserves
finite colimits and compact coefficient actions, so ordinary Ind
extends it to a compact-preserving lift. Any small family of
functors, maps of exact sequences, or coefficient intertwiners lifts simultaneously.
Full faithfulness on $C$, tested also after compact coefficient
actions, identifies their mapping spaces and coefficient Homs.
The compact assertion therefore gives the unrestricted assertion.

Let us prove the last assertion.
The equivalences are compatible with coherent families
of diagrams parametrized by small enriched categories,
as modeled by the diagram modules of
Lemma~\ref{lem:parameter-modules}.
In particular, they preserve the coefficient
intertwining objects.
\end{proof}

\subsection{The universal upper exact sequence}

\begin{lemma}\label{lem:canonical-row-maps}
Let \(s\ge1\) and \(H\in \nPr{s}\).
For an exact sequence $e=(a\xrightarrow{i}b\xrightarrow{p}c)$ in \(H\), put
$e_R=(a\xrightarrow1a\to0)$ and
$e_L=(0\to c\xrightarrow1c)$. There is an exact sequence
$e_R\to e\to e_L$ of exact sequences. At $s\geq2$ its
forward maps and their right adjoints have components
\begin{equation}\label{eq:fnd-upper-components}
\begin{array}{c|ccc}
 &a&b&c\\\hline
 e_R\longrightarrow e&1&i&0\\
 e\longrightarrow e_R&1&r&0\\
 e\longrightarrow e_L&0&p&1\\
 e_L\longrightarrow e&0&j&1.
\end{array}
\end{equation}
The construction is natural in complete coefficient parameters.
\end{lemma}

\begin{proof}
At $s=1$ the forward maps give a pointwise cofiber sequence.
At greater $s$, the identities $ri=1$, $pj=1$, $pi=0$, $rj=0$, and the
triangle identities give the displayed strong comparisons and
invertible mates. Whiskering the central row gives
\[
 i\to i\to0,\qquad r\to r\to0,\qquad
 0\to p\to p,\qquad0\to j\to j.
\]
These degenerate rows have contractible compatible completions,
by the cofiber property at the base and dual uniqueness at each
subsequent stage. This establishes their parameter coherences.
The resulting diagram has vertex matrix
\[
 \begin{pmatrix}a&a&0\\a&b&c\\0&c&c\end{pmatrix}.
\]
Every row and column is the given exact sequence or a degenerate sequence.
Lemma~\ref{lem:exact-row-structure} now proves the assertion,
including exactness of the recursive central row.
\end{proof}

The degenerate-sequence functors
$\sigma_L(x)=(x\xrightarrow1x\to0)$ and
$\sigma_R(x)=(0\to x\xrightarrow1x)$ satisfy
\begin{equation}\label{eq:row-evaluation-adjunctions}
 \sigma_L\dashv\operatorname{ev}_a,\qquad
 \operatorname{ev}_c\dashv\sigma_R.
\end{equation}
These adjoints induce equivalences of coefficient Homs.


\begin{proposition}\label{prop:fnd-upper}
Let \(s\ge1\).
There is an internal exact sequence in the coefficient
category $\nPr s$
\begin{equation}\label{eq:presentable-upper-row}
 \one_s\xrightarrow{I}W_s\xrightarrow{L}\one_s.
\end{equation}
Its central exact sequence evaluated at the distinguished middle object is
the universal exact sequence.
\end{proposition}

\begin{proof}
Let $a\to b\to c$ be the universal exact sequence in \(W_s\).
The maps $I,J:\one_s\to W_s$ select $a,c$. The maps
$R,L:W_s\to\one_s$ classify the degenerate sequences respectively $(1_{s-1}\xrightarrow11_{s-1}\to0)$ and
$(0\to 1_{s-1}\xrightarrow11_{s-1})$ in \(1_s\cong \nPr{(s-1)}\). Under
\eqref{eq:presentable-exact-classifier}, precomposition by $I,J$
is vertex evaluation, and precomposition by $R,L$ is
$\sigma_L,\sigma_R$. The coefficient Hom equivalences
\eqref{eq:row-evaluation-adjunctions} give $I\dashv R$ and
$L\dashv J$, with $RI=1$, $LJ=1$, $LI=0$ and $RJ=0$.

Apply Lemma~\ref{lem:canonical-row-maps} to the universal exact sequence.
Transport its exact sequence $e_R\to e\to e_L$ through the
coefficient equivalence \eqref{eq:presentable-exact-classifier}
with $H=W_s$. Its three functors are $IR,1_{W_s},JL$,
and its two maps are the counit and unit just constructed.
Thus $IR\to1_{W_s}\to JL$ is an exact sequence of module
functors. Lemma~\ref{lem:fnd-parameters} makes it pointwise exact.
Evaluation at $b$ gives the universal exact sequence. This uses neither
membership of $\EqRow_s(W_s)$ in a coefficient category nor a
test only at the three distinguished objects.
\end{proof}

For compact $a$ in a coefficient module, $W_s\odot a$ is compact.
Tensor--Hom adjunction represents a compact local exact sequence by a
map $W_s\odot a\to b$. Tensoring
\eqref{eq:presentable-upper-row} with $a$ and evaluating the central
sequence propagates additivity.

\subsection{Ind-completions}
Restriction to compact objects will determine diagrams on the whole
input. Additivity of that extension needs a separate proof.

\begin{proposition}\label{prop:filtered-extension}
Restriction gives a natural equivalence
\[
 \operatorname{Fun}_{n,\mathrm{filt}}(A,E)
 \simeq\operatorname{Fun}_n^\omega(A,E),
\]
compatible with every coefficient-enriched parameter.
\end{proposition}

\begin{proof}
Given $F_C$ on the right, define its object values by the ordinary
filtered Kan extension
\begin{equation}\label{eq:filtered-kan-values}
 F(a)=\operatorname*{colim}_{(c\to a)\in(A^\omega)_{/a}}F_C(c).
\end{equation}
For $n=1$, this is the ordinary Ind universal property.
Assume $n\geq2$. For compact $c$ and
$b=\operatorname{colim}_jb_j$ with compact $b_j$,
Lemma~\ref{lem:compact-hom-filtered} gives
\[
 \PrHom_A(c,b)\simeq\operatorname*{colim}_j\PrHom_A(c,b_j).
\]
Lemma~\ref{lem:compact-coefficient-homs} identifies its compact
objects and their local coefficient data with the filtered colimit
of the corresponding stagewise data. Compose their $F_C$-images
with $F_Cb_j\to Fb$ and use this source colimit to obtain the local
diagram at $(c,b)$.

For $a=\operatorname{colim}_ia_i$ with compact $a_i$, the action
adjunction gives coefficient equivalences
\begin{equation}\label{eq:filtered-first-hom-limits}
 \PrHom_A(a,b)\simeq\lim_i\PrHom_A(a_i,b),\qquad
 \PrHom_E(Fa,Fb)\simeq\lim_i\PrHom_E(F_Ca_i,Fb).
\end{equation}
The compact-core and coefficient-Hom limit statements of
Lemma~\ref{lem:compact-coefficient-homs} apply at every lower index.
Taking the limit of the previously constructed local diagrams defines
the local diagram at $(a,b)$.

For composition, first restrict the source to a compact $a_i$.
Each finite collection of composable local generators and its
comparison maps descends to common compact target stages. At those
stages the asserted comparison is that of $F_C$. Its associativity
and unit identities are the corresponding identities between maps
of associative algebras. One may check them in each simplicial
degree of their free-algebra resolutions. Each degree uses finite
composable strings, so the same descent applies. The coefficient
limits in \eqref{eq:filtered-first-hom-limits} detect the resulting
identities at arbitrary sources.

The construction commutes with coefficient parameters and is determined by
\eqref{eq:filtered-kan-values} and \eqref{eq:filtered-first-hom-limits}.
Apply Remark~\ref{rem:parameter-detection}.
\end{proof}

\begin{proposition}\label{prop:global-additive-extension}
The filtered extension of $F_C\in\Add_n^\omega(A,E)$ belongs to
$\Add_{n,\mathrm{filt}}(A,E)$.
In particular, 
\[
\Add_{n,\mathrm{filt}}(A,E)\cong \Add_n^\omega(A,E).
\]
\end{proposition}

\begin{proof}
Let $F$ be the filtered extension of
$F_C\in\Add_n^\omega(A,E)$.
It preserves zero since $0\in A^\omega$.
For $n=1$, the universal exact sequence
\[
 x\longrightarrow W_1\odot x\longrightarrow x
\]
is a filtered colimit of such sequences with
compact vertices. Its image under $F$ is therefore
exact. The universal property of $W_1$ gives
\[
 F(g)\simeq F(f)+F(h)
\]
for every cofiber sequence $f\to g\to h$
in a coefficient Hom of $A$.
Applying this to $ir\to1_b\to jp$ shows that
the image of any exact sequence
$a\to b\to c$ is split exact.

Now suppose $n\geq2$.
A compact exact sequence in
$\PrHom_A(a,b)$, for compact $a,b$,
is classified by a map $W_n\odot a\to b$.
The universal upper exact sequence
\eqref{eq:presentable-upper-row}, tensored
with $a$, has compact vertices.
Its image under $F_C$ is exact, and
the classifying map recovers the image
of the prescribed local exact sequence.

For arbitrary $a=\operatorname*{colim}_ia_i$
and $b=\operatorname*{colim}_jb_j$ with
compact $a_i,b_j$, use
\[
 \PrHom_A(a_i,b)
 \simeq\operatorname*{colim}_j\PrHom_A(a_i,b_j),
 \qquad
 \PrHom_A(a,b)
 \simeq\lim_i\PrHom_A(a_i,b).
\]
The compact exact-sequence classifier and
Lemma~\ref{lem:compact-coefficient-homs}
show that the local exactness established
above extends to arbitrary $a,b$.

Finally, $F$ preserves the specified adjunctions
and their coherent data. Since it also preserves
the central exact sequence of every input exact
sequence, it preserves exact sequences by
Definition~\ref{def:genuine}.
\end{proof}
