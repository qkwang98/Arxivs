\section{Estimates for Lorentzian forms}\label{sec: norms}

Let $A^\bullet$ be an algebra of degree $d\geq2$ and satisfy the
K\"ahler package with respect to $K$. For
$\ell_0,\ldots,\ell_{d-2}\in K$, define
\[
 T(x,y)=\deg(\ell_1\cdots\ell_{d-2}xy),\qquad x,y\in A^1.
\]
The degree-zero relations give $T(\ell_0,\ell_0)>0$, and the degree-one
relations say that $T$ is negative definite on the hyperplane
$\{x:T(\ell_0,x)=0\}$. Thus $T$ has signature
$(1,\dim A^1-1)$, where the first entry counts positive eigenvalues.
We call this the \textit{Lorentzian signature}.

Let $W$ be a finite-dimensional real vector space and $T$ be a symmetric bilinear form with the Lorentzian signature. For any $h\in W$ such that $T(h,h)>0$, we can associate a $T$-invariant involution
\[
    \theta_{T,h}\colon W\to W,\quad x\mapsto \frac{2T(h,x)}{T(h,h)}\cdot h-x,
\]
which fixes $h$ and negates $h^\perp$. We define
\[
    G_{T,h}(x,y)=T(\theta_{T,h}(x),y)=\frac{2T(h,x)T(h,y)}{T(h,h)}-T(x,y).
\]
Since $T$ is negative definite on $h^\perp$, for orthogonal decomposition $x=\lambda h+x^\perp$, we have
\[
G_{T,h}(x,x)=T(\lambda h-x^\perp,\lambda h+x^\perp)=\lambda^2T(h,h)-T(x^\perp,x^\perp)\geq 0,
\]
with equality only when $x=0$. Thus $G_{T,h}$ is an inner product on $W$.
\begin{lemma}\label{lem: cauchy-schwarz}
    \[
        T(x,y)^2\leq G_{T,h}(x,x)G_{T,h}(y,y),\quad \forall x,y\in W
    \]
\end{lemma}
\begin{proof}
    \begin{align*}
         T(x,y)^2=& G_{T,h}(\theta_{T,h}x,y)^2\\
         \leq& G_{T,h}(\theta_{T,h}(x),\theta_{T,h}(x))G_{T,h}(y,y)\\
         =&G_{T,h}(x,x)G_{T,h}(y,y),
    \end{align*}
    where the inequality is Cauchy--Schwarz for $G_{T,h}$.
\end{proof}
\begin{lemma}\label{lem: estimate of norms}
    Let $R,S,T$ be symmetric bilinear forms with Lorentzian signature on $W$ satisfying $T=R+S$. Suppose that $h\in W$ satisfies
    \[
        r=R(h,h)>0,\ s=S(h,h)>0.
    \]
    Further assume that there is $u\in W$ with
    \[
        R(h,x)=T(u,x),\quad S(h,x)=T(h-u,x)
    \]
    for all $x\in W$, and that $T(u,u),T(h-u,h-u)\geq 0$. Then we have
    \[
        G_{R,h}(x,x)\leq 5 G_{T,h}(x,x),\quad \forall x\in W.
    \]
\end{lemma}
\begin{proof}
    Set $t=r+s=R(h,h)+S(h,h)=T(h,h)$. We have
    \begin{align*}
        G_{T,h}(u,u)=\frac{2T(u,h)^2}{T(h,h)}-T(u,u)=\frac{2R(h,h)^2}{t}-T(u,u)\leq \frac{2 r^2}{t}.
    \end{align*}
    By \cref{lem: cauchy-schwarz}, we have
    \[
        R(h,x)^2=T(u,x)^2\leq G_{T,h}(u,u)G_{T,h}(x,x)\leq \frac{2r^2}{t}G_{T,h}(x,x).
    \]
    Similarly, we have
    \[
        S(h,x)^2\leq \frac{2s^2}{t}G_{T,h}(x,x) .
    \]
    Since $G_{S,h}$ is positive definite, we have
    \[
        S(x,x)=\frac{2S(h,x)^2}{s}-G_{S,h}(x,x)\leq \frac{2S(h,x)^2}{s}.
    \]
    Therefore
    \begin{align*}
        G_{R,h}(x,x)=&\frac{2R(h,x)^2}{r}-R(x,x)\\
        =&\frac{2R(h,x)^2}{r}+S(x,x)-T(x,x)\\
        \leq&\frac{4r+4s}{t}G_{T,h}(x,x)+\left(G_{T,h}(x,x)-\frac{2T(h,x)^2}{t}\right)\\
        \leq& 5G_{T,h}(x,x).
    \end{align*}
\end{proof}
